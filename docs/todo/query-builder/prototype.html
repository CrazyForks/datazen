实时生成的 SQL
SELECT
  users.id,
  users.name,
  users.email,
  orders.amount
FROM users
INNER JOIN orders
  ON users.id = orders.user_id
WHERE orders.amount > 100
ORDER BY orders.amount DESC
LIMIT 50;
